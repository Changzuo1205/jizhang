                    onClick = {
                        val csv = onGenerateCsv()
                        val sendIntent = Intent().apply {
                            action = Intent.ACTION_SEND
                            putExtra(Intent.EXTRA_TEXT, csv)
                            type = "text/plain"
                        }
                        val shareIntent = Intent.createChooser(sendIntent, "导出 CSV")
                        context.startActivity(shareIntent)
                    }
